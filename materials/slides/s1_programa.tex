\section{Motivación y objetivos del curso}
 
\begin{frame}{Descripción del curso}
\begin{itemize}
	\item En este curso, los estudiantes aprenderán y practicarán \textcolor{blue}{técnicas microeconométricas de punta} empleadas en el trabajo aplicado, con un enfoque en la \textcolor{blue}{estimación de efectos causales}%\pause
	\item Estas técnicas se aplican ampliamente en áreas como Economía Laboral, Organización Industrial, Economía de la Salud y Economía del Desarrollo%\pause
	\item El curso enfatiza \textcolor{blue}{cómo} y \textcolor{blue}{bajo qué condiciones} es apropiado utilizar cada uno de los modelos vistos en clase
\end{itemize}
\end{frame}


\begin{frame}{Resultados de aprendizaje}
\begin{itemize}
    \item[1.] \textbf{Formular} estrategias para abordar problemas de identificación en ejercicios econométricos de inferencia causal %\pause
    
    \item[2.] \textbf{Evaluar} críticamente la validez de una estrategia de identificación en un contexto específico de estimación%\pause
    
    \item[3.] \textbf{Realizar} estimaciones de modelos econométricos utilizando herramientas como Stata, R o Python%\pause
    
    \item[4.] \textbf{Interpretar} correctamente los resultados de un ejercicio econométrico%\pause

     \item[5.] \textbf{Diferenciar} y \textbf{evaluar} la validez interna y externa en un ejercicio de estimación%\pause
    
    \item[6.] \textbf{Comunicar} efectivamente los resultados de una estimación econométrica%\pause
    
    \item[7.] \textbf{Evaluar} la robustez y sensibilidad de los resultados
\end{itemize}
\end{frame}

\section{Contenido temático}

\begin{frame}{¿En qué vamos a trabajar?}	
\begin{itemize}
	\item \textbf{\underline{Econometría}:} estudio unificado de \textcolor{blue}{modelos económicos}, \textcolor{blue}{análisis de datos}, y \textcolor{blue}{estadística matemática} \citep{Hanse2021}. \vspace{0.4cm}	\pause
\begin{itemize}
	\item[1.] \textcolor{red}{\textbf{Modelos económicos}}: construcciones teóricas (simplificaciones) diseñadas para estudiar relaciones económicas
\begin{align*}
y = f(x_1,\ldots,x_K,e_1,\ldots,e_J;\textcolor{brown}{\beta_1},\ldots,\textcolor{brown}{\beta_P})
\end{align*}\vspace{-0.4cm}
	\item $y$ es la \textbf{variable dependiente}
	\item $x_k$,  $k \in \{1,\ldots, K\}$, son \textbf{variables independientes} observadas 
	\item $e_j$, $j \in \{1,\ldots, J\}$, son ``\textbf{perturbaciones}'' o \textbf{factores no observados} 
	\item $\textcolor{brown}{\beta_p}$, $p \in \{1,\ldots, P\}$, son \textbf{parámetros} que median la relación
\end{itemize}	
\end{itemize}
\end{frame}

\begin{frame}{¿En qué vamos a trabajar?}	
\begin{enumerate}
	\item[2.] \textcolor{red}{\textbf{Datos}}: mediciones de las variables de interés.
\begin{align*}
\{(y_i,x_{i1},\ldots,x_{iK}): i = 1, \ldots, N\}
\end{align*}	
\pause
	\item[3.] \textcolor{red}{\textbf{Estadística matemática}}: aplicaciones de teoría de probabilidad. Por ejemplo, si el modelo teórico nos dice que $\pderiv{y}{x_k} = \textcolor{brown}{\beta_k}<0$, quisiéramos estimar $\textcolor{brown}{\beta_k}$:	 
	\begin{align*}
	\textcolor{brown}{\hat{\beta}_k} = h(\yB,\XB)
	\end{align*}
 \begin{itemize}
     \item El estimador es una función de los datos
     \item Los estimadores tienen propiedades estadísticas
 \end{itemize}
\end{enumerate}	
\end{frame}

\begin{frame}{¿En qué vamos a trabajar?}    
\begin{itemize}
    \item Preguntas clave que se abordarán a lo largo del curso:
    \vspace{1em}
    \begin{enumerate}
        \item \textcolor{red}{\textbf{Identificación}:} ¿Bajo qué condiciones podemos hablar de relaciones \textbf{causales}? ¿Qué supuestos teóricos y estadísticos son necesarios para que los parámetros del modelo estén determinados de manera única a partir de la distribución de variables observadas?
        \vspace{1em}%\pause     
        \item \textcolor{red}{\textbf{Métodos de estimación}:} ¿Cómo podemos estimar los parámetros de interés? ¿Qué propiedades estadísticas tienen esos estimadores? 
        \vspace{1em}%\pause        
        \item \textcolor{red}{\textbf{Inferencia estadística}:} Dado que los estimadores son variables aleatorias, ¿qué podemos inferir de su distribución de probabilidad a partir de la muestra?
    \end{enumerate}
\end{itemize}
\end{frame}
